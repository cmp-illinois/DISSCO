73 size_t CalculateTupletLimit() {
74 size_t tuplet_limit = 2; // the whole beat (type 1) is always possible
75 valid_dividers_.clear();
76
77 for (int tuplet_num = 1; tuplet_num <= beat_edus_; ++tuplet_num) {
78 if (beat_edus_ % tuplet_num == 0) {
79 valid_dividers_.push_back(beat_edus_ / tuplet_num);
80 tuplet_limit = static_cast<size_t>(tuplet_num) + 1;
81 }
82 }
83
84 return tuplet_limit;
85 }

94 int DetermineTuplet(int dur) {
95 // valid_dividers_ runs from the longest tuplet note (the whole beat,
96 // type 1) to the shortest, so the first fit is the simplest tuplet
97 for (size_t i = 0; i < valid_dividers_.size(); ++i) {
98 int tuplet_num = beat_edus_ / valid_dividers_[i];
99 if (tuplet_num >= 2 && dur % valid_dividers_[i] == 0) {
100 return tuplet_num;
101 }
102 }
103
104 return -1; // invalid tuplet
105 }

111 void ConstructTupletNames() {
112 tuplet_types_.clear();
113
114 for (size_t i = 0; i < valid_dividers_.size(); ++i) {
115 int tuplet_num = beat_edus_ / valid_dividers_[i];
116 int l = CalculateNearestPow2(tuplet_num);
117 string t = "\\tuplet ";
118 t += Note::int_to_str(tuplet_num) + "/" + Note::int_to_str(l) + "{ ";
119 tuplet_types_[tuplet_num] = t;
120 }
121 }

168 static int DiscreteLog2(int num) {
169 if (!IsPowerOf2(num)) {
170 return -1;
171 }
172
173 int pow = 0;
174 while (num > 1) {
175 num /= 2;
176 ++pow;
177 }
178
179 return pow;
180 }

188 static bool IsPowerOf2(int num) {
189 return num > 0 && (num & (num - 1)) == 0;
190 }

199 static int CalculateNearestPow2(int num) {
200 int multiple_of_2 = 1;
201
202 while (multiple_of_2 <= num) {
203 multiple_of_2 *= 2;
204 }
205
206 return multiple_of_2 / 2;
207 }

216 static int Power(int base, int pow) {
