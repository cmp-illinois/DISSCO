57static int runCmod(int parameterCount, char **parameterList) {
58 // Rubin Du 2024: Installed custom signal handler to print stack trace on segfault
59 signal(SIGSEGV, segfaultHandler);
60
61 time_t startTime;
62 time(&startTime);
63
64 //Determine settings.
65 cout << endl;
66 cout << "=========================SETTINGS==========================" << endl;
67
68 //Determine the project path.
69 string path;
70 if(parameterCount >= 2)
71 path = parameterList[1];
72 if(path == "--help" || path == "-help" || path == "help") {
73 cout << "Usage: cmod <project.dissco> Builds the specified project." << endl;
74 //cout << " cmod <path> <process-offset=0> <process-count=1>" << endl;
75 //cout << " Renders a specific mask of sounds." << endl;
76 cout << " cmod help Displays this help." << endl;
77 return 0;
78 }
79
80 if (path.empty()) {
81 throw CmodError(CmodError::Kind::Project, "No project file was specified.",
82 "Command line", "Run cmod <path-to-project.dissco>.");
83 }
84
85 const filesystem::path projectPath(path);
86 filesystem::path workingDirectory = projectPath.parent_path();
87 if (workingDirectory.empty())
88 workingDirectory = ".";
89
90 string workingPath = workingDirectory.string();
91 cout << "Working in path: " << workingPath << endl;
92
93 //Determine the project name.
94 string projectName = projectPath.stem().string();
95
96 //Create the piece!
97 Piece piece(workingPath, projectName);
98 //delete outputFile; //Sever
99
100 time_t endTime;
101 time(&endTime);
102
103 int seconds = static_cast<int>(difftime(endTime, startTime));
104 int hr = seconds / 3600;
105 int min = (seconds % 3600) / 60;
106 int sec = seconds % 60;
107 printf("Computation Time: %02d:%02d:%02d.\n", hr, min, sec);
108
109
110 return 0;
111
112}

114int main(int parameterCount, char **parameterList) {
115 const char* project = parameterCount >= 2 ? parameterList[1] : "(not specified)";
116 try {
117 return runCmod(parameterCount, parameterList);
118 } catch (const CmodError& error) {
119 error.report(cerr, project);
120 return error.exitCode();
121 } catch (const std::exception& error) {
122 CmodError failure(CmodError::Kind::Internal, error.what(), "Building project",
123 "Report this problem to the DISSCO developers with the project, seed, and this diagnostic.");
124 failure.report(cerr, project);
125 return failure.exitCode();
126 } catch (...) {
127 CmodError failure(CmodError::Kind::Internal, "An unexpected failure occurred.",
128 "Building project",
129 "Report this problem to the DISSCO developers with the project, seed, and this diagnostic.");
130 failure.report(cerr, project);
131 return failure.exitCode();
132 }
133}
