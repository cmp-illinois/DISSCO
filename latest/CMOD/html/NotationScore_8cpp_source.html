16// Chooses the clef for the notes of a built section from their average
17// pitch: treble if it is not below middle C (pitch 48), else bass. Returns
18// "" for a section with only rests.
19static string ChooseClef(const list<Note*>& section_flat) {
20 int avePitchNum = 0;
21 int pitchSum = 0;
22 for (list<Note*>::const_iterator iter = section_flat.begin();
23 iter != section_flat.end();
24 ++iter) {
25 if ((*iter)->is_real_note()) {
26 avePitchNum = avePitchNum + (*iter)->getPitchNum();
27 pitchSum = pitchSum + 1;
28 }
29 }
30 if (pitchSum == 0) {
31 return "";
32 }
33 return (avePitchNum / pitchSum >= 48) ? "treble" : "bass";
34}

36// Writes a note value, given as a fraction of a whole note, as a LilyPond
37// duration: a power-of-two value, plain or with up to two dots ("4", "4.",
38// "8.."). Returns "" for a value that cannot be written so, such as a
39// triplet quarter (1/6).
40static string LilyPondDuration(Ratio value) {
41 int dots;
42 if (value.Num() == 1) {
43 dots = 0;
44 } else if (value.Num() == 3) {
45 dots = 1;
46 } else if (value.Num() == 7) {
47 dots = 2;
48 } else {
49 return "";
50 }
51 // The undotted value, 1/undotted, is 2^dots / Den: a whole note or shorter
52 const int undotted = value.Den() >> dots;
53 if (!TimeSignature::IsPowerOf2(value.Den()) || undotted < 1) {
54 return "";
55 }
56 return to_string(undotted) + string(static_cast<size_t>(dots), '.');
57}

59// Gets the tempo mark of a tempo, written as it was entered (\tempo 4 = 60,
60// \tempo 4. = 80). LilyPond writes only plain or dotted beats and whole
61// numbers of beats per minute, so a beat it cannot write (such as a triplet
62// quarter) is restated in the beat of the time signature, which is the same
63// tempo, and beats per minute that are not a whole number are rounded to
64// the nearest one.
65static string TempoMark(Tempo tempo) {
66 string beat = LilyPondDuration(tempo.getTempoBeat());
67 Ratio beats_per_minute = tempo.getTempoBeatsPerMinute();
68 if (beat.empty()) {
69 beat = LilyPondDuration(tempo.getTimeSignatureBeat());
70 beats_per_minute = tempo.getTimeSignatureBeatsPerMinute();
71 }
72 const long long rounded = std::max(1LL, std::llround(beats_per_minute.To<double>()));
73 return "\\tempo " + beat + " = " + to_string(rounded);
74}

76// Gets how many whole notes a tempo plays per minute. Two tempos are the
77// same when a whole note lasts as long in both, such as quarter = 60 in 4/4
78// and eighth = 120 in 6/8.
79static Rational<long long> WholeNotesPerMinute(Tempo tempo) {
80 Ratio beats_per_minute = tempo.getTempoBeatsPerMinute();
81 Ratio beat = tempo.getTempoBeat();
82 return Rational<long long>(static_cast<long long>(beats_per_minute.Num()) * beat.Num(),
83 static_cast<long long>(beats_per_minute.Den()) * beat.Den());
84}

131// In a polymetric score (staves in different meters at the same time),
132// notates a staff's own time signature wherever it has a barline and
133// another staff notates a time signature, even where its meter goes on,
134// so a player does not take the other staff's change for its own. The
135// staves are built; bars are placed by their written length in whole
136// notes from the start of the score, as LilyPond places them.
137static void NotateTimeSignaturesTogether(vector< vector<Section> >& staves) {
138 struct Bar {
139 Ratio start;
140 string time_signature;
141 Section* section;
142 size_t index; // in its section
143 bool notated; // its time signature is notated at its start
144 };
145 vector< vector<Bar> > bars(staves.size());
146 for (size_t i = 0; i < staves.size(); i++) {
147 Ratio start(0);
148 for (vector<Section>::iterator section = staves[i].begin();
149 section != staves[i].end();
150 ++section) {
151 vector<string> time_signatures = section->GetBarTimeSignatures();
152 vector<bool> marks = section->GetBarTimeSignatureMarks();
153 for (size_t k = 0; k < time_signatures.size(); k++) {
154 Bar bar = {start, time_signatures[k], &*section, k, k < marks.size() && marks[k]};
155 bars[i].push_back(bar);
156 start += Ratio(time_signatures[k]);
157 }
158 }
159 }

161 // For each bar of one staff, the bar of another staff in force where it
162 // starts (every staff starts with the score) tells whether their meters
163 // differ there, and whether the other staff has a barline where this
164 // one notates a time signature
165 bool polymetric = false;
166 vector<Bar*> to_notate;
167 for (size_t i = 0; i < bars.size(); i++) {
168 for (size_t j = 0; j < bars.size(); j++) {
169 if (i == j || bars[j].empty()) {
170 continue;
171 }
172 size_t in_force = 0;
173 for (size_t k = 0; k < bars[i].size(); k++) {
174 Bar& bar = bars[i][k];
175 while (in_force + 1 < bars[j].size() && bars[j][in_force + 1].start <= bar.start) {
176 ++in_force;
177 }
178 Bar& other = bars[j][in_force];
179 polymetric = polymetric || other.time_signature != bar.time_signature;
180 if (bar.notated && other.start == bar.start) {
181 to_notate.push_back(&other);
182 }
183 }
184 }
185 }
186 if (!polymetric) {
187 return;
188 }
189 for (size_t k = 0; k < to_notate.size(); k++) {
190 to_notate[k]->section->NotateBarTimeSignature(to_notate[k]->index);
191 }
192}

212// Rejects a score in which tempo sections (notes not yet inserted) with
213// different tempos overlap in time on different staves (on one staff,
214// MergeOverlappingSections rejects them): the staves of a score share one
215// tempo, so the staves that sound at the same time must have the same one
216// (WholeNotesPerMinute). The sections of each staff are already merged, so
217// they overlap as the score places them (StartOffsetEDUs). A tempo change
218// after the notes of the earlier tempo end is fine.
219static void CheckSimultaneousTempos(vector< vector<Section> >& staves) {
220 struct TempoSection {
221 Section* section;
222 int staff;
223 int last_note_end;
224 Rational<long long> whole_notes_per_minute;
225 };
226 vector<TempoSection> sections;
227 for (size_t i = 0; i < staves.size(); i++) {
228 for (vector<Section>::iterator iter = staves[i].begin();
229 iter != staves[i].end();
230 ++iter) {
231 TempoSection tempo_section = {&*iter, static_cast<int>(i), iter->GetLastNoteEnd(),
232 WholeNotesPerMinute(iter->GetTimeSignature().tempo_)};
233 sections.push_back(tempo_section);
234 }
235 }
236 std::stable_sort(sections.begin(), sections.end(),
237 [](const TempoSection& a, const TempoSection& b) {
238 return a.section->GetStartTimeGlobal() < b.section->GetStartTimeGlobal();
239 });
240
241 for (size_t later = 1; later < sections.size(); later++) {
242 for (size_t earlier = 0; earlier < later; earlier++) {
243 TempoSection& first = sections[earlier];
244 TempoSection& second = sections[later];
245 if (first.whole_notes_per_minute == second.whole_notes_per_minute) {
246 continue;
247 }
248 if (first.last_note_end <= StartOffsetEDUs(*first.section, *second.section)) {
249 continue;
250 }
251 throw CmodError(CmodError::Kind::Project,
252 "Two tempo sections overlap in time but differ in tempo. Staves that sound "
253 "at the same time must share one tempo in the score.",
254 "Score output, sections starting at " + DescribeSection(*first.section) +
255 " on staff " + to_string(first.staff) + " and at " +
256 DescribeSection(*second.section) + " on staff " +
257 to_string(second.staff),
258 "Give events that sound at the same time the same tempo (the same length "
259 "of a whole note, such as quarter = 60 in 4/4 and eighth = 120 in 6/8), "
260 "start the later one after the earlier one's notes end, or disable score "
261 "output.");
262 }
263 }
264}

266// Merges the sections of a staff (in time order, notes not yet inserted)
267// that overlap in time: where the notes of one still sound when the next
268// starts, the next section's notes go onto the bars of the first, which
269// continue until the merged music ends. That needs one bar grid for both:
270// the same tempo, time signature and EDU Per Beat. The next section starts
271// at the nearest edu of the first one's grid (StartOffsetEDUs), and its
272// notes keep their offsets from that start. Sections that overlap without
273// sharing a grid cannot be notated on one staff.
274static void MergeOverlappingSections(vector<Section>& sections, int staff) {
275 size_t earlier = 0;
276 while (earlier + 1 < sections.size()) {
277 Section& first = sections[earlier];
278 Section& second = sections[earlier + 1];
279 const int offset_edus = StartOffsetEDUs(first, second);
280 if (first.GetLastNoteEnd() <= offset_edus) {
281 ++earlier;
282 continue;
283 }
284
285 TimeSignature first_ts = first.GetTimeSignature();
286 TimeSignature second_ts = second.GetTimeSignature();
287 if (first_ts.time_signature_ != second_ts.time_signature_ ||
288 first_ts.beat_edus_ != second_ts.beat_edus_ ||
289 first_ts.tempo_.getTimeSignatureBeatsPerMinute() !=
290 second_ts.tempo_.getTimeSignatureBeatsPerMinute()) {
291 throw CmodError(CmodError::Kind::Project,
292 "Two tempo sections overlap on one staff but differ in tempo, time signature "
293 "or EDU Per Beat, so they cannot share its bars.",
294 "Score output, staff " + to_string(staff) + ", sections starting at " +
295 DescribeSection(first) + " and at " + DescribeSection(second),
296 "Give the overlapping events the same Tempo, Time Signature and EDU Per "
297 "Beat, start the later one after the earlier one's notes end, or put their "
298 "notes on different staves.");
299 }
300
301 first.TakeNotes(second, offset_edus);
302 sections.erase(sections.begin() + static_cast<vector<Section>::difference_type>(earlier + 1));
303 }
304}

306// Fills the time a staff has no section of its own with sections of rests,
307// so that it covers the whole score in step with the other staves. Where a
308// section of the score (timeline, in time order) starts, the staff takes it
309// as rests unless it has a section of its own starting there, or the notes
310// of its section in force are still sounding there. So the rests come
311// before its first section and after the notes of a section have ended;
312// a section of its own that overlaps other staves' sections is kept.
313static void FillWithRests(vector<Section>& sections,
314 const vector<TimeSignature>& timeline) {
315 for (vector<TimeSignature>::const_iterator ts = timeline.begin();
316 ts != timeline.end();
317 ++ts) {
318 Section rests(*ts);
319 // The first section of the staff that does not start before this one
320 vector<Section>::iterator next = sections.begin();
321 while (next != sections.end() && *next < *ts) {
322 ++next;
323 }
324 // Less than one edu of rests would be left before it
325 if (next != sections.end() &&
326 rests.CalculateEDUsFromSecondsInTempo(next->GetStartTimeGlobal() -
327 ts->start_time_global_) == 0) {
328 continue;
329 }
330 if (next != sections.begin()) {
331 Section& previous = *(next - 1);
332 int previous_edus = previous.CalculateEDUsFromSecondsInTempo(
333 ts->start_time_global_ - previous.GetStartTimeGlobal());
334 if (previous_edus == 0 || previous.GetLastNoteEnd() > previous_edus) {
335 continue;
336 }
337 }
338 sections.insert(next, rests);
339 }
340}

481void NotationScore::Build() {
482 if (!is_built_) {
483 for (map<int, int>::const_iterator staff = out_of_range_staffs_.begin();
484 staff != out_of_range_staffs_.end();
485 ++staff) {
486 cout << "Warning: Staff Number " << staff->first << " does not exist in this score (staves 0 to "
487 << staffSum - 1 << "), so " << staff->second << (staff->second == 1 ? " note was" : " notes were")
488 << " placed on staff " << StaffIndex(staff->first) << ". Suggestion: Use a Staff Number from 0 to "
489 << staffSum - 1 << (is_grand ? "." : ", or raise Number of Staff in the project properties.") << endl;
490 }
491
492 // Sections that overlap on a staff share the bars of the earlier one,
493 // and staves that sound at the same time share one tempo; then each
494 // section's notes go into its bars
495 for (int i = 0; i < staffSum; i++) {
496 MergeOverlappingSections(score_staff[i], i);
497 }
498 CheckSimultaneousTempos(score_staff);
499 for (int i = 0; i < staffSum; i++) {
500 for (vector<Section>::iterator iter = score_staff[i].begin();
501 iter != score_staff[i].end();
502 ++iter) {
503 iter->InsertAddedNotes();
504 }
505 }
506
507 // Every section of the score, in time order; at the same start time
508 // the section of the lowest staff comes first
509 vector<TimeSignature> timeline;
510 for (int i = 0; i < staffSum; i++) {
511 for (vector<Section>::iterator iter = score_staff[i].begin();
512 iter != score_staff[i].end();
513 ++iter) {
514 if (iter->GetLastNoteEnd() == 0) {
515 throw CmodError(CmodError::Kind::Project,
516 "No notes with a notatable duration remain in this score section.",
517 "Score output, staff " + to_string(i) + ", section starting at " +
518 to_string(iter->GetStartTimeGlobal()) + " seconds",
519 "Check note durations and EDU settings. Give notes a positive duration that can be represented in the score, or disable score output.");
520 }
521 timeline.push_back(iter->GetTimeSignature());
522 }
523 }
524 if (timeline.empty()) {
525 throw CmodError(CmodError::Kind::Project,
526 "No note events were generated for the score.",
527 "Score output",
528 "Add note events to the project, or disable score output.");
529 }
530 std::stable_sort(timeline.begin(), timeline.end(),
531 [](const TimeSignature& a, const TimeSignature& b) {
532 return a.start_time_global_ < b.start_time_global_;
533 });
534
535 // The score starts at the start of the piece: the silence before the
536 // first section is a section of rests in its tempo and meter, which
537 // ends in a cap bar where it does not fill whole bars
538 Tempo piece_start = timeline.front().tempo_;
539 if (piece_start.convertSecondsToEDUs(timeline.front().start_time_global_) > 0) {
540 piece_start.setStartTime(0);
541 timeline.insert(timeline.begin(), TimeSignature(piece_start));
542 }
543
544 // A staff with no notes, or with no notes yet (such as before the first
545 // section), is rests in step with the others
546 for (int i = 0; i < staffSum; i++) {
547 FillWithRests(score_staff[i], timeline);
548 }
549
550 // All staves end together, at the end of the bar that holds the last
551 // note of the score: a staff that ends sooner gets rests up to there
552 float score_end = 0;
553 for (int i = 0; i < staffSum; i++) {
554 Section& last = score_staff[i].back();
555 score_end = std::max(score_end, last.GetStartTimeGlobal() +
556 last.CalculateSecondsFromEDUsInTempo(EndOfLastBar(last)));
557 }
558 vector<int> last_section_edus(score_staff.size(), -1);
559 for (int i = 0; i < staffSum; i++) {
560 Section& last = score_staff[i].back();
561 int edus = last.CalculateEDUsFromSecondsInTempo(score_end - last.GetStartTimeGlobal());
562 if (edus > EndOfLastBar(last)) {
563 last_section_edus[i] = edus;
564 }
565 }
566
567 // A section that several staves carry is warned about once
568 vector< pair<float, float> > warned_extensions;
569
570 for(int i=0 ; i<staffSum; i++){
571 try {
572 // Since all tempos are registered, calculate their start times
573 // in terms of the previous tempo's EDU's
574 vector<Section>::iterator iter = score_staff[i].begin();
575 vector<Section>::iterator next = score_staff[i].begin() + 1;
576 string previous_time_signature;
577 bool first_section = true;
578 // Each staff starts with no dynamic and carries it across its sections
579 string staff_loudness;
580
581 while (next != score_staff[i].end()) {
582 float dur_seconds = next->GetStartTimeGlobal() - iter->GetStartTimeGlobal();
583 iter->SetDurationEDUS(iter->CalculateEDUsFromSecondsInTempo(dur_seconds));
584
585 string current_time_signature = iter->GetTimeSignature().time_signature_;
586 bool print_time_signature = first_section || current_time_signature != previous_time_signature;
587 iter->Build(print_time_signature, staff_loudness);
588 WarnCapExtension(*iter, warned_extensions);
589
590 first_section = false;
591 // The next section compares with the meter in force after any cap bar
592 previous_time_signature = iter->GetEndingTimeSignature().time_signature_;
593 ++iter; ++next;
594 }
595
596 iter->SetDurationEDUS(last_section_edus[i]);
597 string current_time_signature = iter->GetTimeSignature().time_signature_;
598 bool print_time_signature = first_section || current_time_signature != previous_time_signature;
599 iter->Build(print_time_signature, staff_loudness);
600 WarnCapExtension(*iter, warned_extensions);
601 } catch (CmodError& error) {
602 error.addContext("Score output, staff " + to_string(i));
603 throw;
604 }
