230ModifierDetailsDialog::partialRowConstraint() const
231{
232 ModifierUiPolicy::PartialRowConstraint constraint;
233 ProjectManager* projectManager = Inst::get_project_manager();
234 if (!projectManager || !projectManager->get_curr_project()) {
235 constraint.maximumRows = 1;
236 constraint.explanation = tr(
237 "No open project is available; one placeholder partial row is shown.");
238 return constraint;
239 }
240
241 const QList<SpectrumEvent>& spectra = projectManager->spectrumevents();
242 QList<const SpectrumEvent*> candidates;
243 QStringList resolutionIssues;
244 QSet<QString> seenNames;
245 bool bottomScope = false;
246 bool runtimeDependentPath = false;
247
248 enum class PackageTypeKind { Spectrum, Other, RuntimeDependent };
249 const auto classifyPackageType = [](const QString& value) {
250 const QString type = value.trimmed();
251 bool validInteger = false;
252 const int integerType = type.toInt(&validInteger);
253 if (validInteger) {
254 return integerType == static_cast<int>(sound)
255 ? PackageTypeKind::Spectrum : PackageTypeKind::Other;
256 }
257
258 bool validNumber = false;
259 const double numericType = type.toDouble(&validNumber);
260 if (validNumber && std::isfinite(numericType)
261 && std::floor(numericType) == numericType) {
262 return numericType == static_cast<double>(sound)
263 ? PackageTypeKind::Spectrum : PackageTypeKind::Other;
264 }
265
266 if (type == QStringLiteral("Spectrum"))
267 return PackageTypeKind::Spectrum;
268 return PackageTypeKind::RuntimeDependent;
269 };
270
271 const auto addSpectrumCandidates = [&](const QString& name) {
272 if (name.isEmpty()) {
273 resolutionIssues.append(tr(
274 "A Spectrum package has an empty event name."));
275 runtimeDependentPath = true;
276 return;
277 }
278 if (seenNames.contains(name))
279 return;
280 seenNames.insert(name);
281
282 QList<const SpectrumEvent*> matches;
283 for (const SpectrumEvent& spectrum : spectra) {
284 if (spectrum.name == name)
285 matches.append(&spectrum);
286 }
287 if (matches.isEmpty()) {
288 resolutionIssues.append(tr(
289 "Referenced Spectrum \"%1\" does not exist.").arg(name));
290 runtimeDependentPath = true;
291 return;
292 }
293 if (matches.size() > 1) {
294 resolutionIssues.append(tr(
295 "Spectrum name \"%1\" is duplicated, so the reference is "
296 "ambiguous.").arg(name));
297 runtimeDependentPath = true;
298 }
299 for (const SpectrumEvent* match : matches) {
300 if (!candidates.contains(match))
301 candidates.append(match);
302 }
303 };
304
305 if (m_eventType == bottom) {
306 if (m_eventIndex
307 >= static_cast<unsigned>(projectManager->bottomevents().size())) {
308 constraint.maximumRows = 0;
309 constraint.explanation = tr(
310 "The current Bottom event is no longer available. Reopen its "
311 "modifier before configuring partial rows.");
312 return constraint;
313 }
314 bottomScope = true;
315 const HEvent& bottomEvent =
316 projectManager->bottomevents()[static_cast<int>(m_eventIndex)].event;
317 for (const Layer& layer : bottomEvent.event_layers) {
318 for (const Package& package : layer.discrete_packages) {
319 const PackageTypeKind kind =
320 classifyPackageType(package.event_type);
321 if (kind == PackageTypeKind::Other)
322 continue;
323 if (kind == PackageTypeKind::RuntimeDependent) {
324 runtimeDependentPath = true;
325 resolutionIssues.append(tr(
326 "The child type for package \"%1\" is evaluated at "
327 "runtime, so its Spectrum status cannot be guaranteed.")
328 .arg(package.event_name));
329 }
330 // EventName is a literal key in CMOD. Do not trim or otherwise
331 // normalize it here, or the editor could validate a reference
332 // that runtime lookup will reject.
333 addSpectrumCandidates(package.event_name);
334 }
335 }
336 } else {
337 for (const SpectrumEvent& spectrum : spectra) {
338 if (seenNames.contains(spectrum.name)) {
339 runtimeDependentPath = true;
340 resolutionIssues.append(tr(
341 "Spectrum name \"%1\" is duplicated in the project.")
342 .arg(spectrum.name));
343 }
344 seenNames.insert(spectrum.name);
345 candidates.append(&spectrum);
346 }
347 }
348
349 if (candidates.isEmpty()) {
350 constraint.maximumRows = 0;
351 constraint.explanation = bottomScope
352 ? tr("This Bottom has no statically resolved Spectrum candidate. "
353 "PARTIAL settings do not affect note-only children; a runtime "
354 "child expression cannot be capped in the editor.")
355 : tr("No reachable Spectrum can be resolved for this modifier. "
356 "One placeholder row is shown without a Spectrum-derived limit.");
357 if (!resolutionIssues.isEmpty())
358 constraint.explanation += tr("\nReview: %1")
359 .arg(resolutionIssues.join(QStringLiteral(" ")));
360 return constraint;
361 }
362
363 bool exactMaximum = !runtimeDependentPath;
364 int maximum = 1;
365 for (const SpectrumEvent* spectrum : candidates) {
366 const ModifierUiPolicy::SpectrumPartialCount count =
367 ModifierUiPolicy::spectrumPartialCount(*spectrum);
368 maximum = std::max(maximum, count.count);
369 exactMaximum = exactMaximum && count.exact;
370
371 if (!count.generated && count.exact) {
372 int configuredPartials = 0;
373 for (const QString& partial : spectrum->spectrum.partials) {
374 if (!partial.trimmed().isEmpty())
375 ++configuredPartials;
376 }
377 if (configuredPartials != count.count) {
378 resolutionIssues.append(tr(
379 "Spectrum \"%1\" declares %2 partials but contains %3 "
380 "configured partial envelopes.")
381 .arg(spectrum->name)
382 .arg(count.count)
383 .arg(configuredPartials));
384 }
385 if (!spectrum->generate_spectrum.trimmed().isEmpty()) {
386 resolutionIssues.append(tr(
387 "Spectrum \"%1\" has GenerateSpectrum text but no function "
388 "element; CMOD will use its explicit partial list.")
389 .arg(spectrum->name));
390 }
391 } else if (!count.exact) {
392 resolutionIssues.append(tr(
393 "Spectrum \"%1\" determines NumberOfPartials at runtime.")
394 .arg(spectrum->name));
395 }
396 }
397
398 constraint.suggestedRows = maximum;
399 constraint.maximumRows = exactMaximum ? maximum : 0;
400 if (!exactMaximum) {
401 constraint.explanation = bottomScope
402 ? tr("At least one Spectrum referenced by this Bottom has a runtime-"
403 "determined or inconsistent partial count. %1 rows are suggested; "
404 "CMOD uses the actual count at runtime.").arg(maximum)
405 : tr("At least one project Spectrum has a runtime-determined or "
406 "inconsistent partial count. %1 rows are suggested; inherited "
407 "modifiers use each sound's actual count at runtime.").arg(maximum);
408 } else if (bottomScope && candidates.size() == 1) {
409 constraint.explanation = tr(
410 "This Bottom references Spectrum \"%1\", which declares up to %2 "
411 "partials. You can configure at most %2 rows; CMOD may use fewer "
412 "at high base frequencies.")
413 .arg(candidates.front()->name)
414 .arg(maximum);
415 } else if (bottomScope) {
416 constraint.explanation = tr(
417 "This Bottom can choose %1 Spectra; the largest has %2 partials. "
418 "You can configure at most %2 rows; smaller spectra ignore later rows.")
419 .arg(candidates.size())
420 .arg(maximum);
421 } else {
422 constraint.explanation = tr(
423 "For inherited modifiers, the project-wide safe maximum covers %1 "
424 "Spectra; the largest declares up to %2 partials. You can configure "
425 "at most %2 rows.")
426 .arg(candidates.size())
427 .arg(maximum);
428 }
429 if (!resolutionIssues.isEmpty())
430 constraint.explanation += tr("\nReview: %1")
431 .arg(resolutionIssues.join(QStringLiteral(" ")));
432 return constraint;
433}

435void ModifierDetailsDialog::accept()
436{
437 const bool applyByPartial = (m_applyCombo->currentIndex() == 1);
438 for (const FieldWidgets& widgets : m_fields) {
439 if (!ModifierUiPolicy::fieldEnabled(
440 static_cast<int>(m_modifier.type),
441 static_cast<int>(widgets.field), applyByPartial)) {
442 continue;
443 }
444
445 const QString value = widgets.edit->text().trimmed();
446 if (widgets.field == PartialResult) {
447 const QString error = PartialModifierFormat::validationError(
448 static_cast<int>(m_modifier.type), value);
449 if (!error.isEmpty()) {
450 QMessageBox::warning(
451 this, tr("Invalid partial parameters"), error);
452 widgets.edit->setFocus();
453 return;
454 }
455 continue;
456 }
457 if (value.isEmpty()
458 || value.compare(
459 QStringLiteral("N/A"), Qt::CaseInsensitive) == 0) {
460 QMessageBox::warning(
461 this, tr("Missing modifier parameter"),
462 tr("Enter or generate every parameter shown for this "
463 "modifier before saving."));
464 widgets.edit->setFocus();
465 return;
466 }
467 if ((widgets.field == Spread
468 || widgets.field == Direction
469 || widgets.field == Velocity)
470 && !widgets.edit->hasAcceptableInput()) {
471 QMessageBox::warning(
472 this, tr("Invalid modifier parameter"),
473 tr("Enter a valid number for every Detune parameter."));
474 widgets.edit->setFocus();
475 return;
476 }
477 if (widgets.field == Direction) {
478 bool validDirection = false;
479 const double direction = value.toDouble(&validDirection);
480 if (!validDirection || !std::isfinite(direction)
481 || direction == 0.0) {
482 QMessageBox::warning(
483 this, tr("Invalid Detune direction"),
484 tr("Enter a negative value to detune or a positive "
485 "value to tune. Direction cannot be zero."));
486 widgets.edit->setFocus();
487 return;
488 }
489 }
490 }
491
492 m_modifier.applyhow_flag = applyByPartial;
493 for (const FieldWidgets& widgets : m_fields) {
494 if (widgets.field == Direction
495 && ModifierUiPolicy::fieldEnabled(
496 static_cast<int>(m_modifier.type),
497 static_cast<int>(widgets.field), applyByPartial)) {
498 const double direction = widgets.edit->text().toDouble();
499 setValue(widgets.field,
500 direction < 0.0 ? QStringLiteral("-1")
501 : QStringLiteral("1"));
502 } else {
503 setValue(widgets.field, widgets.edit->text());
504 }
505 }
506 QDialog::accept();
