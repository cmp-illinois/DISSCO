11void writeDomNode(QXmlStreamWriter& writer, const QDomNode& node)
12{
13 if (node.isElement()) {
14 const QDomElement element = node.toElement();
15 writer.writeStartElement(element.tagName());
16 const QDomNamedNodeMap attributes = element.attributes();
17 for (int i = 0; i < attributes.count(); ++i) {
18 const QDomAttr attribute = attributes.item(i).toAttr();
19 writer.writeAttribute(attribute.name(), attribute.value());
20 }
21 for (QDomNode child = element.firstChild(); !child.isNull();
22 child = child.nextSibling()) {
23 writeDomNode(writer, child);
24 }
25 writer.writeEndElement();
26 } else if (node.isCDATASection()) {
27 writer.writeCDATA(node.toCDATASection().data());
28 } else if (node.isText()) {
29 const QString text = node.toText().data();
30 if (!text.trimmed().isEmpty())
31 writer.writeCharacters(text);
32 }
33}

174bool ProjectXmlWriter::updateProjectSeed(const QString& filePath,
175 const QString& seed)
176{
177 QFile input(filePath);
178 if (!input.open(QIODevice::ReadOnly | QIODevice::Text))
179 return false;
180
181 QDomDocument document;
182 if (!document.setContent(&input))
183 return false;
184 input.close();
185
186 QDomElement seedElement = document.documentElement()
187 .firstChildElement(QStringLiteral("ProjectConfiguration"))
188 .firstChildElement(QStringLiteral("Seed"));
189 if (seedElement.isNull())
190 return false;
191
192 while (!seedElement.firstChild().isNull())
193 seedElement.removeChild(seedElement.firstChild());
194 seedElement.appendChild(document.createTextNode(seed));
195
196 QSaveFile output(filePath);
197 if (!output.open(QIODevice::WriteOnly | QIODevice::Text))
198 return false;
199
200 QTextStream stream(&output);
201 stream << document.toString();
202 stream.flush();
203 if (stream.status() != QTextStream::Ok) {
204 output.cancelWriting();
205 return false;
206 }
207 return output.commit();
