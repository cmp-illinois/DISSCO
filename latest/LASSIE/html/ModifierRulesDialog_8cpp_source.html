369bool ModifierRulesDialog::hasAmbiguousContext(
370 const ModifierChanceRule& candidate,
371 int ignoredIndex) const
372{
373 for (int index = 0; index < m_rules.size(); ++index) {
374 if (index == ignoredIndex
375 || m_rules[index].conditions.size()
376 != candidate.conditions.size()) {
377 continue;
378 }
379
380 bool canOverlap = true;
381 for (const ModifierCondition& candidateCondition
382 : candidate.conditions) {
383 for (const ModifierCondition& existingCondition
384 : m_rules[index].conditions) {
385 if (candidateCondition.modifier_id
386 == existingCondition.modifier_id
387 && candidateCondition.required_on
388 != existingCondition.required_on) {
389 canOverlap = false;
390 break;
391 }
392 }
393 if (!canOverlap)
394 break;
395 }
396 if (canOverlap)
397 return true;
398 }
399 return false;
400}
