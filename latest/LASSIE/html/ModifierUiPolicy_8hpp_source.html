31inline bool staticIntegerValue(const QString& source, int* result = nullptr)
32{
33 const QString value = source.trimmed();
34 bool validInteger = false;
35 const int integer = value.toInt(&validInteger);
36 if (validInteger) {
37 if (result)
38 *result = integer;
39 return true;
40 }
41
42 bool validNumber = false;
43 const double number = value.toDouble(&validNumber);
44 if (!validNumber
45 || !std::isfinite(number)
46 || std::floor(number) != number
47 || number < std::numeric_limits<int>::min()
48 || number > std::numeric_limits<int>::max()) {
49 return false;
50 }
51 if (result)
52 *result = static_cast<int>(number);
53 return true;
54}

98inline SpectrumPartialCount spectrumPartialCount(const SpectrumEvent& spectrum)
99{
100 QXmlStreamReader generatedReader(
101 QStringLiteral("<root>") + spectrum.generate_spectrum
102 + QStringLiteral("</root>"));
103 bool hasGeneratedElement = false;
104 bool sawWrapperElement = false;
105 while (!generatedReader.atEnd()) {
106 generatedReader.readNext();
107 if (!generatedReader.isStartElement())
108 continue;
109 if (!sawWrapperElement) {
110 sawWrapperElement = true;
111 continue;
112 }
113 hasGeneratedElement = true;
114 }
115 if (!generatedReader.hasError() && hasGeneratedElement)
116 return {generatedSpectrumPartialCount, true, true};
117
118 const int listedCount = std::max(
119 1, static_cast<int>(spectrum.spectrum.partials.size()));
120
121 int declaredCount = 0;
122 if (staticIntegerValue(spectrum.num_partials, &declaredCount)
123 && declaredCount > 0) {
124 return {declaredCount, true, false};
125 }
126
127 // A function-valued NumberOfPartials is evaluated by CMOD. The explicit
128 // list is still the best editor suggestion, but it is not a safe limit.
129 return {listedCount, false, false};
130}

132inline int editorRowMaximum(const PartialRowConstraint& constraint,
133 int savedRows)
134{
135 const int preservedRows = std::max(1, savedRows);
136 if (constraint.maximumRows > 0)
137 return std::max(constraint.maximumRows, preservedRows);
138 // Runtime-valued NumberOfPartials has no truthful finite UI cap. The row
139 // count control is read-only and grows one row per Add action, so using
140 // QSpinBox's full integer range does not trigger a bulk allocation.
141 return std::numeric_limits<int>::max();
142}
