266void PartialModifierDialog::setPartialRowCount(int count)
267{
268 const int previousCount = m_activeRowCount;
269 while (m_rows.size() < count)
270 addPartialRow(m_rows.size(), PartialModifierFormat::Values{});
271
272 if (count < previousCount) {
273 for (int index = count; index < previousCount; ++index) {
274 PartialRow& row = m_rows[index];
275 row.probability->setText(QStringLiteral("N/A"));
276 row.magnitude->setText(QStringLiteral("N/A"));
277 row.width->setText(QStringLiteral("N/A"));
278 row.rate->setText(QStringLiteral("N/A"));
279 }
280 }
281
282 m_activeRowCount = count;
283 for (int index = 0; index < m_rows.size(); ++index)
284 m_rows[index].group->setVisible(index < m_activeRowCount);
285 updateCountControls();
286 updatePreview();
287
288 if (count > previousCount && count <= m_rows.size()) {
289 PartialRow& row = m_rows[count - 1];
290 row.probability->setFocus();
291 if (m_scrollArea)
292 m_scrollArea->ensureWidgetVisible(row.group);
293 }

296QString PartialModifierDialog::resultString() const
297{
298 QVector<PartialModifierFormat::Values> values;
299 values.reserve(m_activeRowCount);
300 for (int index = 0; index < m_activeRowCount; ++index) {
301 const PartialRow& row = m_rows[index];
302 PartialModifierFormat::Values value;
303 value.probability = row.probability->text();
304 value.magnitude = row.magnitude->text();
305 value.width = row.width->text();
306 value.rate = row.rate->text();
307 values.append(value);
308 }
309 return PartialModifierFormat::serialize(values);

312void PartialModifierDialog::accept()
313{
314 if (!ModifierUiPolicy::rowCountAllowed(
315 m_constraint, m_activeRowCount, m_savedRowCount)) {
316 QMessageBox::warning(
317 this, tr("Too many partial rows"),
318 tr("This configuration cannot grow beyond %1 rows. Reduce the row "
319 "count or cancel to preserve the original value.")
320 .arg(std::max(m_constraint.maximumRows, m_savedRowCount)));
321 return;
322 }
323
324 const QString error = PartialModifierFormat::validationError(
325 m_modifierType, resultString());
326 if (!error.isEmpty()) {
327 QMessageBox::warning(this, tr("Invalid partial parameters"), error);
328 return;
329 }

333void PartialModifierDialog::updateCountControls()
334{
335 if (!m_rowCountSpin)
336 return;
337
338 const bool hasExactLimit = m_constraint.maximumRows > 0;
339 const bool aboveCurrentLimit =
340 hasExactLimit && m_activeRowCount > m_constraint.maximumRows;
341 const bool belowSuggestion =
342 m_activeRowCount < m_suggestedPartialCount;
343
344 QString status = m_statusContext;
345 if (!status.isEmpty())
346 status += QLatin1Char('\n');
347 status += tr("Showing %1 partial row(s).").arg(m_activeRowCount);
348 m_statusLabel->setText(status);
349
350 m_addPartialButton->setEnabled(
351 m_activeRowCount < m_rowCountSpin->maximum()
352 && (!hasExactLimit
353 || m_activeRowCount < m_constraint.maximumRows));
354 m_removePartialButton->setEnabled(m_activeRowCount > 1);
355
356 if (aboveCurrentLimit) {
357 m_countWarningLabel->setText(tr(
358 "This saved modifier has %1 rows, while the current Spectrum limit "
359 "is %2. The extra rows are preserved but CMOD ignores them for these "
360 "sounds; no additional rows can be added.")
361 .arg(m_activeRowCount)
362 .arg(m_constraint.maximumRows));
363 m_countWarningLabel->setVisible(true);
364 } else if (belowSuggestion) {
365 m_countWarningLabel->setText(tr(
366 "Partials after row %1 will not receive this modifier.")
367 .arg(m_activeRowCount));
368 m_countWarningLabel->setVisible(true);
369 } else {
370 m_countWarningLabel->setVisible(false);
371 }
