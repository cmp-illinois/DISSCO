15void writeText(QXmlStreamWriter& writer, const QString& text, bool compact)
16{
17 // Keep actual whitespace inside parameter values without putting literal
18 // line breaks or tabs into the one-line entry that receives the result.
19 // CR and NBSP also need references in the preview: XML and QTextEdit
20 // otherwise normalize them to LF and a regular space, respectively.
21 qsizetype start = 0;
22 for (qsizetype i = 0; i < text.size(); ++i) {
23 const QChar c = text[i];
24 if (c != '\r' && c != QChar(0x00a0)
25 && !(compact && (c == '\n' || c == '\t')))
26 continue;
27 writer.writeCharacters(text.mid(start, i - start));
28 writer.writeEntityReference(QStringLiteral("#%1").arg(static_cast<uint>(c.unicode())));
29 start = i + 1;
30 }
31 writer.writeCharacters(text.mid(start));
32}

44void writeNode(QXmlStreamWriter& writer, const QDomNode& node,
45 bool compact, bool preserveSpace = false)
46{
47 if (node.isElement()) {

50 if (space == QStringLiteral("preserve")) preserveSpace = true;
51 else if (space == QStringLiteral("default")) preserveSpace = false;
52
53 bool hasElements = false;
54 bool hasText = false;
55 for (QDomNode child = node.firstChild(); !child.isNull(); child = child.nextSibling()) {
56 hasElements |= child.isElement();
57 hasText |= child.isCDATASection()
58 || (child.isText() && !isXmlWhitespace(child.nodeValue()));
59 }
60

62 writer.writeStartElement(element.tagName());
63 const QDomNamedNodeMap attributes = element.attributes();
64 for (int i = 0; i < attributes.count(); ++i) {
65 const QDomAttr attribute = attributes.item(i).toAttr();
66 writer.writeAttribute(attribute.name(), attribute.value());
67 }
68
69 // A List can mix literal values and nested functions. Adding layout
70 // whitespace there would change its text, so keep that subtree inline.
71 writer.setAutoFormatting(parentFormatting && !hasText && !preserveSpace);
72 for (QDomNode child = node.firstChild(); !child.isNull(); child = child.nextSibling()) {
73 if (hasElements && !hasText && !preserveSpace && child.isText()
74 && !child.isCDATASection() && isXmlWhitespace(child.nodeValue()))
75 continue;
76 writeNode(writer, child, compact, preserveSpace);
77 }
78 writer.writeEndElement();

80 } else if (node.isText() || node.isCDATASection()) {

82 } else if (node.isComment()) {
