95 || envelopes.isNull()) {

105 !envelope.isNull();

107 const QString value = elementInnerXml(envelope).trimmed();
108 envelopeValues.append(value.isEmpty() ? QStringLiteral("N/A") : value);
109 }
110
111 if (envelopeValues.size() % 4 != 0) {
112 if (warning) {
113 *warning = translated(
114 "The existing Partials function does not contain four envelopes per partial; missing values were filled with N/A.");
115 }
116 while (envelopeValues.size() % 4 != 0)
117 envelopeValues.append(QStringLiteral("N/A"));
118 }
119
120 for (int i = 0; i < envelopeValues.size(); i += 4) {
121 Values row;
122 row.probability = envelopeValues.at(i);
123 row.magnitude = envelopeValues.at(i + 1);
124 row.width = envelopeValues.at(i + 2);
125 row.rate = envelopeValues.at(i + 3);

147QString PartialModifierFormat::validationError(int modifierType,
148 const QString& source)
149{
150 QString warning;
151 const QVector<Values> values = parse(source, &warning);
152 if (!warning.isEmpty())
153 return warning;
154 if (values.isEmpty())
155 return translated("Configure at least one partial row.");
156
157 for (int row = 0; row < values.size(); ++row) {
158 const Values& value = values[row];
159 if (!hasConfiguredValue(value.probability))
160 continue;
161
162 const auto require = [&](const QString& effectValue,
163 const char* effectName) -> QString {
164 return hasConfiguredValue(effectValue)
165 ? QString()
166 : missingEffectError(row, translated(effectName));
167 };
168
169 QString error;
170 switch (modifierType) {
171 case 0: // Tremolo
172 case 1: // Vibrato
173 case 7: // Phase Modulation
174 error = require(value.magnitude, "Magnitude");
175 if (error.isEmpty())
176 error = require(value.rate, "Rate");
177 break;
178 case 2: // Glissando
179 case 3: // Detune
180 case 6: // Wave Type
181 error = require(value.magnitude, "Magnitude");
182 break;
183 case 4: // Amplitude Transient
184 case 5: // Frequency Transient
185 error = require(value.magnitude, "Magnitude");
186 if (error.isEmpty())
187 error = require(value.width, "Width");
188 if (error.isEmpty())
189 error = require(value.rate, "Rate");
190 break;
191 default:
192 return translated("The selected modifier type is not supported.");
193 }
194 if (!error.isEmpty())
195 return error;
196 }
197 return {};
