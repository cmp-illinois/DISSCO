248void EnvelopeLibraryWindow::copyEnvelope()
249{
250 if (!activeProject) return;
251 const auto rows = envelopeLibrary->selectionModel()->selectedRows();
252 if (rows.isEmpty()) return;
253 auto* item = refModel->itemFromIndex(rows.first());
254 auto* entry = static_cast<EnvelopeLibraryEntry*>(
255 item->data(Qt::UserRole).value<void*>());
256 if (!entry) return;
257
258 EnvelopeClipboard copied;
259 for (auto* node = entry->head; node;) {
260 EnvelopeClipboardNode value{node->x, node->y};
261 if (node->rightSeg) {
262 value.type = node->rightSeg->segmentType;
263 value.property = node->rightSeg->segmentProperty;
264 }
265 copied.nodes.append(value);
266 node = node->rightSeg ? node->rightSeg->rightNode : nullptr;
267 }
268 ProjectClipboard::copy(Inst::get_project_manager()->get_curr_project(),
269 std::move(copied), tr("Envelope %1").arg(entry->number));
270}

272void EnvelopeLibraryWindow::pasteEnvelope()
273{
274 if (!activeProject) return;
275 auto* pm = Inst::get_project_manager();
276 const auto* copied =
277 ProjectClipboard::get<EnvelopeClipboard>(pm->get_curr_project());
278 if (!copied || copied->nodes.size() < 2) return;
279
280 auto* last = pm->envlibentries();
281 while (last && last->next) last = last->next;
282 auto* entry = new EnvelopeLibraryEntry(last ? last->number + 1 : 1);
283 auto* node = entry->head;
284 for (qsizetype index = 0; index < copied->nodes.size(); ++index) {
285 const auto& value = copied->nodes[index];
286 node->x = value.x;
287 node->y = value.y;
288 if (index + 1 == copied->nodes.size()) break;
289
290 // Reuse the default envelope's two endpoints and add any others.
291 if (!node->rightSeg) {
292 node->rightSeg = new EnvLibEntrySeg;
293 node->rightSeg->leftNode = node;
294 node->rightSeg->rightNode = new EnvLibEntryNode(0.0, 0.0);
295 node->rightSeg->rightNode->leftSeg = node->rightSeg;
296 }
297 node->rightSeg->segmentType = value.type;
298 node->rightSeg->segmentProperty = value.property;
299 node = node->rightSeg->rightNode;
300 }
301
302 entry->prev = last;
303 if (last) last->next = entry;
304 else pm->envlibentries() = entry;
305
306 auto* item = new QStandardItem(entry->getNumberString());
307 item->setData(QVariant::fromValue<void*>(entry), Qt::UserRole);
308 refModel->appendRow(item);
309 envelopeLibrary->setCurrentIndex(refModel->index(refModel->rowCount() - 1, 0));
310 MUtilities::modified();
311}
