35void validateEffectParameters(const Modifier& modifier,
36 int oneBasedPosition,
37 QStringList& diagnostics)
38{
39 const auto require = [&](const QString& value, const QString& field) {
40 if (!hasConfiguredValue(value)) {
41 diagnostics.append(
42 QObject::tr("Modifier %1 is missing %2.")
43 .arg(oneBasedPosition)
44 .arg(field));
45 }
46 };
47 const auto requireRange = [&](const QString& value,
48 const QString& field,
49 double minimum,
50 double maximum) {
51 if (!hasConfiguredValue(value)) {
52 require(value, field);
53 return;
54 }
55 bool valid = false;
56 const double number = value.trimmed().toDouble(&valid);
57 if (!valid || !std::isfinite(number)
58 || number < minimum || number > maximum) {
59 diagnostics.append(
60 QObject::tr("Modifier %1: %2 must be between %3 and %4.")
61 .arg(oneBasedPosition)
62 .arg(field)
63 .arg(minimum)
64 .arg(maximum));
65 }
66 };
67 const auto requireDirection = [&](const QString& value) {
68 const QString field = QObject::tr("Detune Direction");
69 if (!hasConfiguredValue(value)) {
70 require(value, field);
71 return;
72 }
73 bool valid = false;
74 const double direction = value.trimmed().toDouble(&valid);
75 if (!valid || !std::isfinite(direction) || direction == 0.0) {
76 diagnostics.append(
77 QObject::tr("Modifier %1: Detune Direction must be a "
78 "finite, non-zero number.")
79 .arg(oneBasedPosition));
80 }
81 };
82
83 if (modifier.applyhow_flag) {
84 const QString error = PartialModifierFormat::validationError(
85 static_cast<int>(modifier.type),
86 modifier.partialresult_string);
87 if (!error.isEmpty()) {
88 diagnostics.append(
89 QObject::tr("Modifier %1: %2")
90 .arg(oneBasedPosition)
91 .arg(error));
92 }
93 return;
94 }
95
96 switch (modifier.type) {
97 case 0: // Tremolo
98 case 1: // Vibrato
99 case 7: // Phase Modulation
100 require(modifier.amplitude, QObject::tr("Magnitude"));
101 require(modifier.rate, QObject::tr("Rate"));
102 break;
103 case 2: // Glissando
104 case 6: // Wave Type
105 require(modifier.amplitude, QObject::tr("Magnitude"));
106 break;
107 case 3: // Detune
108 requireRange(modifier.detune_spread,
109 QObject::tr("Detune Spread"), 0.0, 1.0);
110 requireDirection(modifier.detune_direction);
111 requireRange(modifier.detune_velocity,
112 QObject::tr("Detune Velocity"), -1.0, 1.0);
113 break;
114 case 4: // Amplitude Transient
115 case 5: // Frequency Transient
116 require(modifier.amplitude, QObject::tr("Magnitude"));
117 require(modifier.rate, QObject::tr("Rate"));
118 require(modifier.width, QObject::tr("Width"));
119 break;
120 default:
121 diagnostics.append(
122 QObject::tr("Modifier %1 has an unknown type.")
123 .arg(oneBasedPosition));
124 break;
125 }
126}
