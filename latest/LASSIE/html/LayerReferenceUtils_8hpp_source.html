40inline bool referenceExists(const Package& package) {
41 auto* pm = Inst::get_project_manager();
42 if (!pm->get_curr_project()) return false;
43 bool validType = false;
44 const int type = package.event_type.toInt(&validType);
45 if (!validType) return false;
46 const auto containsName = [&](const auto& list) {
47 return std::any_of(list.cbegin(), list.cend(), [&](const auto& event) {
48 return event.name == package.event_name;
49 });
50 };
51 switch (type) {
52 case top: return pm->topevent().name == package.event_name;
53 case high: return containsName(pm->highevents());
54 case mid: return containsName(pm->midevents());
55 case low: return containsName(pm->lowevents());
56 case bottom:
57 return std::any_of(pm->bottomevents().cbegin(), pm->bottomevents().cend(),
58 [&](const BottomEvent& event) { return event.event.name == package.event_name; });
59 case sound: return containsName(pm->spectrumevents());
60 case note: return containsName(pm->noteevents());
61 case env: return containsName(pm->envelopeevents());
62 case sieve: return containsName(pm->sieveevents());
63 case spa: return containsName(pm->spaevents());
64 case pattern: return containsName(pm->patternevents());
65 case reverb: return containsName(pm->reverbevents());
66 case filter: return containsName(pm->filterevents());
67 default: return false;
68 }
69}

99inline Assoc collect(const QString& typeNumStr, const QString& name) {
100 Assoc refs;
101 ProjectManager* pm = Inst::get_project_manager();
102
103 auto scan = [&](HEvent& h, Eventtype ownerType, int ownerIndex) {
104 for (int li = 0; li < h.event_layers.size(); ++li) {
105 const QList<Package>& pkgs = h.event_layers[li].discrete_packages;
106 QList<int> hits;
107 for (int pi = 0; pi < pkgs.size(); ++pi) {
108 if (pkgs[pi].event_type == typeNumStr &&
109 pkgs[pi].event_name == name) {
110 hits.append(pi);
111 }
112 }
113 if (!hits.isEmpty())
114 refs.append(qMakePair(OwnerKey{ownerType, ownerIndex, li}, hits));
115 }
116 };
117
118 scan(pm->topevent(), top, -1);
119 for (int i = 0; i < pm->highevents().size(); ++i)
120 scan(pm->highevents()[i], high, i);
121 for (int i = 0; i < pm->midevents().size(); ++i)
122 scan(pm->midevents()[i], mid, i);
123 for (int i = 0; i < pm->lowevents().size(); ++i)
124 scan(pm->lowevents()[i], low, i);
125 for (int i = 0; i < pm->bottomevents().size(); ++i)
126 scan(pm->bottomevents()[i].event, bottom, i);
127
128 return refs;
129}

135inline HEvent* resolveOwner(const OwnerKey& k) {
136 ProjectManager* pm = Inst::get_project_manager();
137 switch (k.ownerType) {
138 case top: return &pm->topevent();
139 case high: return (k.ownerIndex < pm->highevents().size()) ? &pm->highevents()[k.ownerIndex] : nullptr;
140 case mid: return (k.ownerIndex < pm->midevents().size()) ? &pm->midevents()[k.ownerIndex] : nullptr;
141 case low: return (k.ownerIndex < pm->lowevents().size()) ? &pm->lowevents()[k.ownerIndex] : nullptr;
142 case bottom: return (k.ownerIndex < pm->bottomevents().size()) ? &pm->bottomevents()[k.ownerIndex].event : nullptr;
143 default: return nullptr;
144 }

152inline void applyRename(const Assoc& refs, const QString& newName) {
153 for (const auto& entry : refs) {
154 HEvent* h = resolveOwner(entry.first);
155 if (!h || entry.first.layerIndex >= h->event_layers.size()) continue;
156 QList<Package>& pkgs = h->event_layers[entry.first.layerIndex].discrete_packages;
157 for (int pi : entry.second)
158 if (pi >= 0 && pi < pkgs.size()) pkgs[pi].event_name = newName;
159 }
160}

168inline void applyDelete(const Assoc& refs) {
169 for (const auto& entry : refs) {
170 HEvent* h = resolveOwner(entry.first);
171 if (!h || entry.first.layerIndex >= h->event_layers.size()) continue;
172 QList<Package>& pkgs = h->event_layers[entry.first.layerIndex].discrete_packages;
173 QList<int> desc = entry.second;
174 std::sort(desc.begin(), desc.end(), std::greater<int>());
175 for (int pi : desc)
176 if (pi >= 0 && pi < pkgs.size()) pkgs.removeAt(pi);
177 }
178}
