24bool isInteractiveInput(const QEvent* event)
25{
26 const QEvent::Type type = event->type();
27 if (type == QEvent::MouseMove) {

57QMargins effectiveTextMargins(const QLineEdit* edit)
58{
59 QMargins margins = edit->textMargins();
60 const QRect editRect = edit->rect();
61 int leftSideWidgetWidth = 0;
62 int rightSideWidgetWidth = 0;
63
64 const auto childWidgets = edit->findChildren<QWidget*>(
65 QString(), Qt::FindDirectChildrenOnly);
66 for (const QWidget* child : childWidgets) {
67 if (!child->isVisibleTo(edit))
68 continue;
69
70 const QRect geometry = child->geometry().intersected(editRect);
71 if (geometry.isEmpty())
72 continue;
73
74 const int distanceFromLeft = geometry.left() - editRect.left();
75 const int distanceFromRight = editRect.right() - geometry.right();
76 if (distanceFromLeft <= distanceFromRight) {
77 leftSideWidgetWidth = qMax(
78 leftSideWidgetWidth,
79 geometry.right() - editRect.left() + 1);
80 } else {
81 rightSideWidgetWidth = qMax(
82 rightSideWidgetWidth,
83 editRect.right() - geometry.left() + 1);
84 }
85 }
86
87 margins.setLeft(margins.left() + leftSideWidgetWidth);
88 margins.setRight(margins.right() + rightSideWidgetWidth);
89 return margins;
90}

100bool TextOverflowDisplayPolicy::eventFilter(QObject* watched, QEvent* event)
101{
102 const QEvent::Type type = event->type();
103 const bool displayMayNeedReset = type == QEvent::Polish
104 || type == QEvent::FocusOut || type == QEvent::Show
105 || type == QEvent::Resize;
106 const bool interactiveInput = isInteractiveInput(event);
107 if (!displayMayNeedReset && !interactiveInput) {
108 return false;
109 }
110
111 auto* edit = qobject_cast<QLineEdit*>(watched);
112 if (!edit)
113 return false;
114
115 if (interactiveInput) {
116 markInteractiveOperation(edit);
117 return false;
118 }
119
120 if (type == QEvent::Polish) {
121 registerLineEdit(edit);
122 if (!edit->hasFocus())
123 showBeginning(edit);
124 return false;
125 }
126
127 if (type == QEvent::Show || type == QEvent::Resize) {
128 if (!edit->hasFocus())
129 scheduleShowBeginning(edit);
130 return false;
131 }
132
133 const auto reason = static_cast<QFocusEvent*>(event)->reason();
134 if (reason != Qt::PopupFocusReason
135 && reason != Qt::ActiveWindowFocusReason) {
136 scheduleShowBeginning(edit, false, true);
137 }
138
139 return false;
140}

142void TextOverflowDisplayPolicy::registerLineEdit(QLineEdit* edit)
143{
144 if (edit->property(kRegisteredProperty).toBool())
145 return;
146
147 edit->setProperty(kRegisteredProperty, true);
148 connect(edit, &QLineEdit::textEdited, edit, [edit]() {
149 markInteractiveOperation(edit);
150 });
151 connect(edit, &QLineEdit::textChanged, edit, [edit]() {
152 scheduleShowBeginning(edit, true);
153 });
154 connect(edit, &QLineEdit::cursorPositionChanged, edit, [edit]() {
155 // setText(existingText) still moves the cursor to the end but emits no
156 // textChanged signal. User input is marked by the event filter, so the
157 // focused case can be handled without disrupting cursor navigation.
158 scheduleShowBeginning(edit, true);
159 });
160 connect(edit, &QLineEdit::returnPressed, edit, [edit]() {
161 scheduleShowBeginningAfterEditing(edit);
162 });
163}

165void TextOverflowDisplayPolicy::markInteractiveOperation(QLineEdit* edit)
166{
167 const qulonglong generation =
168 edit->property(kInteractiveGenerationProperty).toULongLong() + 1;
169 edit->setProperty(kInteractiveGenerationProperty, generation);
170 edit->setProperty(kInteractiveEditProperty, true);
171 // Keep the marker through every queued callback created by this input,
172 // regardless of the order of textEdited and textChanged. Clear it on the
173 // following event-loop turn.
174 QTimer::singleShot(0, edit, [edit, generation]() {
175 QTimer::singleShot(0, edit, [edit, generation]() {
176 if (edit->property(kInteractiveGenerationProperty).toULongLong()
177 == generation) {
178 edit->setProperty(kInteractiveEditProperty, false);
179 }
180 });
181 });

200void TextOverflowDisplayPolicy::scheduleShowBeginningAfterEditing(

202{

205 // returnPressed is emitted inside the key event. Wait until the
206 // interactive-operation marker has completed before restoring the view.
207 QTimer::singleShot(0, edit, [edit, generation]() {

211 // move that interaction's cursor.

233bool TextOverflowDisplayPolicy::textOverflows(const QLineEdit* edit)
234{
235 QStyleOptionFrame option = styleOptionFor(edit);
236 QRect textRect = edit->style()->subElementRect(
237 QStyle::SE_LineEditContents, &option, edit);
238 textRect = textRect.marginsRemoved(effectiveTextMargins(edit));
239 const int availableWidth = textRect.width()
240 - 2 * kLineEditHorizontalMargin;
241
242 QTextLayout layout(edit->displayText(), edit->font());
243 layout.beginLayout();
244 const QTextLine line = layout.createLine();
245 layout.endLayout();
246 const int usedWidth = line.isValid()
