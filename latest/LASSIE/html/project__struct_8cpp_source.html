30 inline QString readInner(QXmlStreamReader& r) {
31 QString result;
32 QXmlStreamWriter w(&result);
33 bool hasChildElement = false;
34 QString textBuffer;
35 int depth = 0;
36 while (!r.atEnd()) {
37 r.readNext();
38 switch (r.tokenType()) {
39 case QXmlStreamReader::StartElement: {
40 hasChildElement = true;
41 w.writeStartElement(r.name().toString());
42 const auto attrs = r.attributes();
43 for (const auto& a : attrs)
44 w.writeAttribute(a.name().toString(), a.value().toString());
45 ++depth;
46 break;
47 }
48 case QXmlStreamReader::EndElement:
49 if (depth == 0)
50 return hasChildElement ? result : textBuffer;
51 w.writeEndElement();
52 --depth;
53 break;
54 case QXmlStreamReader::Characters: {
55 QString text = r.text().toString();
56 if (hasChildElement) {
57 if (!text.trimmed().isEmpty()) {
58 w.writeCharacters(text.trimmed());
59 }
60 } else
61 textBuffer += text;
62 break;
63 }
64 default:
65 break;
66 }
67 }
68 return hasChildElement ? result : textBuffer;
69 }

166 inline ModifierChanceRule parseModifierException(QXmlStreamReader& r) {
167 // r at <Exception>; attributes are read before consuming its children.
168 ModifierChanceRule rule;
169 rule.on_chance =
170 r.attributes().value(QStringLiteral("onChance")).toString();
171
172 while (r.readNextStartElement()) {
173 if (r.name() != QStringView(u"When")) {
174 r.skipCurrentElement();
175 continue;
176 }
177
178 const auto attributes = r.attributes();
179 ModifierCondition condition;
180 condition.modifier_id =
181 attributes.value(QStringLiteral("modifierId")).toString().trimmed();
182
183 bool validState = false;
184 bool requiredOn = true;
185 validState = parseModifierState(
186 attributes.value(QStringLiteral("state")).toString(),
187 requiredOn);
188 condition.required_on = requiredOn;
189
190 // Malformed predicates are ignored rather than being silently
191 // interpreted as an OFF dependency.
192 if (!condition.modifier_id.isEmpty() && validState)
193 rule.conditions.append(condition);
194 r.skipCurrentElement();
195 }
196 return rule;

199 inline bool parseModifierUsage(QXmlStreamReader& r, Modifier& modifier) {
200 // r at <Usage>.
201 const auto attributes = r.attributes();
202 const QString id =
203 attributes.value(QStringLiteral("id")).toString().trimmed();
204 const bool hasDefaultOn =
205 attributes.hasAttribute(QStringLiteral("defaultOn"));
206 const QString defaultOn =
207 attributes.value(QStringLiteral("defaultOn")).toString();
208 if (!id.isEmpty())
209 modifier.instance_id = id;
210 if (hasDefaultOn)
211 modifier.default_on_chance = defaultOn;
212
213 modifier.rules.clear();
214 while (r.readNextStartElement()) {
215 if (r.name() != QStringView(u"Exceptions")) {
216 r.skipCurrentElement();
217 continue;
218 }
219
220 while (r.readNextStartElement()) {
221 if (r.name() == QStringView(u"Exception"))
222 modifier.rules.append(parseModifierException(r));
223 else
224 r.skipCurrentElement();
225 }
226 }
227
228 return ModifierUsageImportPolicy::hasCompleteMetadata(
229 id, hasDefaultOn, defaultOn);

232 inline Modifier parseModifier(QXmlStreamReader& r) {
233 Modifier modifier;
234 bool hasCompleteUsageMetadata = false;
235 while (r.readNextStartElement()) {
236 const QString tag = r.name().toString();
237
238 if (tag == QStringLiteral("Type")) {
239 bool valid = false;
240 const unsigned type = readInner(r).trimmed().toUInt(&valid);
241 if (valid)
242 modifier.type = type;
243 } else if (tag == QStringLiteral("ApplyHow")) {
244 const QString value = readInner(r).trimmed();
245 bool valid = false;
246 const int applyHow = value.toInt(&valid);
247 if (valid) {
248 modifier.applyhow_flag = (applyHow != 0);
249 } else if (value.compare(QStringLiteral("PARTIAL"),
250 Qt::CaseInsensitive) == 0) {
251 modifier.applyhow_flag = true;
252 } else if (value.compare(QStringLiteral("SOUND"),
253 Qt::CaseInsensitive) == 0) {
254 modifier.applyhow_flag = false;
255 }
256 } else if (tag == QStringLiteral("Probability")) {
257 // Consumed only for one-way import of pre-Modifier-Usage files.
258 r.skipCurrentElement();
259 } else if (tag == QStringLiteral("Amplitude")) {
260 modifier.amplitude = readInner(r);
261 } else if (tag == QStringLiteral("Rate")) {
262 modifier.rate = readInner(r);
263 } else if (tag == QStringLiteral("Width")) {
264 modifier.width = readInner(r);
265 } else if (tag == QStringLiteral("DetuneSpread")) {
266 modifier.detune_spread = readInner(r);
267 } else if (tag == QStringLiteral("DetuneDirection")) {
268 modifier.detune_direction = readInner(r);
269 } else if (tag == QStringLiteral("DetuneVelocity")) {
270 modifier.detune_velocity = readInner(r);
271 } else if (tag == QStringLiteral("GroupName")) {
272 // Legacy group membership has no equivalent in the new model.
273 r.skipCurrentElement();
274 } else if (tag == QStringLiteral("PartialResultString")) {
275 modifier.partialresult_string = readInner(r);
276 } else if (tag == QStringLiteral("Usage")) {
277 hasCompleteUsageMetadata = parseModifierUsage(r, modifier);
278 } else {
279 // A future field must not shift the interpretation of any
280 // legacy sibling.
281 r.skipCurrentElement();
282 }
283 }
284
285 if (modifier.instance_id.trimmed().isEmpty()) {
286 modifier.instance_id =
287 QUuid::createUuid().toString(QUuid::WithoutBraces);
288 }
289 modifier.usage_metadata_needs_review = !hasCompleteUsageMetadata;
290 return modifier;
291 }

397 inline void parseExtraInfo(QXmlStreamReader& r, ExtraInfo& info) {
398 // Phase was added after Loudness. Parse by tag rather than position so
399 // projects written before <Phase> existed keep all following fields
400 // aligned. Unknown future fields are ignored safely as well.
401 info.phase = QStringLiteral("0");
402 bool modifierUsageMarkerSupported = false;
403 info.modifier_sampling_scope = ModifierSamplingScope::PerSound;
404 QSet<QString> requiredFields{
405 QStringLiteral("FrequencyInfo"), QStringLiteral("Loudness"),
406 QStringLiteral("Spatialization"), QStringLiteral("Reverb"),
407 QStringLiteral("Modifiers")
408 };
409 while (r.readNextStartElement()) {
410 const QString tag = r.name().toString();
411 requiredFields.remove(tag);
412
413 if (tag == QStringLiteral("FrequencyInfo")) {
414 info.freq_info.freq_flag = nextRequiredChildInner(
415 r, QStringView(u"FrequencyFlag")).toUInt();
416 info.freq_info.continuum_flag = nextRequiredChildInner(
417 r, QStringView(u"FrequencyContinuumFlag")).toUInt();
418 info.freq_info.entry_1 = nextRequiredChildInner(
419 r, QStringView(u"FrequencyEntry1"));
420 info.freq_info.entry_2 = nextRequiredChildInner(
421 r, QStringView(u"FrequencyEntry2"));
422 consumeRest(r);
423 } else if (tag == QStringLiteral("Loudness")) {
424 info.loudness = readInner(r);
425 } else if (tag == QStringLiteral("Phase")) {
426 const QString phase = readInner(r);
427 info.phase = phase.trimmed().isEmpty() ? QStringLiteral("0") : phase;
428 } else if (tag == QStringLiteral("Spatialization")) {
429 info.spa = readInner(r);
430 } else if (tag == QStringLiteral("Reverb")) {
431 info.reverb = readInner(r);
432 } else if (tag == QStringLiteral("Filter")) {
433 info.filter = readInner(r);
434 } else if (tag == QStringLiteral("ModifierGroup")) {
435 // Consume the old selector without keeping a second model.
436 r.skipCurrentElement();
437 } else if (tag == QStringLiteral("ModifierUsage")) {
438 const QString version = r.attributes()
439 .value(QStringLiteral("version"))
440 .toString()
441 .trimmed();
442 const QString scope = r.attributes()
443 .value(QStringLiteral("samplingScope"))
444 .toString()
445 .trimmed()
446 .toLower();
447 const bool scopeSupported =
448 scope == QStringLiteral("per-sound")
449 || scope == QStringLiteral("per-bottom");
450 modifierUsageMarkerSupported =
451 version == QStringLiteral("1") && scopeSupported;
452 info.modifier_sampling_scope =
453 scope == QStringLiteral("per-bottom")
454 ? ModifierSamplingScope::PerBottom
455 : ModifierSamplingScope::PerSound;
456 r.skipCurrentElement();
457 } else if (tag == QStringLiteral("Modifiers")) {
458 parseModifiers(r, info.modifiers);
459 } else {
460 r.skipCurrentElement();
461 }
462 }
463
464 if (!r.hasError() && !requiredFields.isEmpty()) {
465 QStringList missingFields = requiredFields.values();
466 missingFields.sort();
467 r.raiseError(QStringLiteral(
468 "<ExtraInfo> is missing required element(s): %1.")
469 .arg(missingFields.join(QStringLiteral(", "))));
470 }
471 info.modifier_usage_needs_review = !modifierUsageMarkerSupported;
472 }

533bool Project::parseEvent(QXmlStreamReader& r, Eventtype* parsedType) {
534 // r at <Event> StartElement (with orderInPalette attribute).
535 QString orderinpalette = r.attributes().value("orderInPalette").toString();
536
537 // First child is <EventType>, whose text content is the integer type.
538 if (!r.readNextStartElement()) {
539 r.raiseError(QStringLiteral("<Event> is missing <EventType>."));
540 return false;
541 }
542 if (r.name() != QStringView(u"EventType")) {
543 r.raiseError(QStringLiteral("Expected <EventType> as the first child of <Event>."));
544 return false;
545 }
546
547 bool validType = false;
548 int typeInt = QtParser::readInner(r).toInt(&validType);
549 if (!validType || typeInt < static_cast<int>(top)
550 || typeInt > static_cast<int>(spec)) {
551 r.raiseError(QStringLiteral("<EventType> contains an invalid event type."));
552 return false;
553 }
554 Eventtype type = (Eventtype)typeInt;
555 if (type == folder || type == mea || type == spec) {
556 r.raiseError(QStringLiteral("<EventType> contains an unsupported event type."));
557 return false;
558 }
559
560 switch (type) {
561 case top:
562 case high:
563 case mid:
564 case low: {
565 HEvent eh;
566 eh.orderinpalette = orderinpalette;
567 eh.type = type;
568 QtParser::parseHEventChildren(r, eh);
569 if (!QtParser::hasValidEventName(r, eh.name))
570 return false;
571 qDebug() << "parsed" << eh.type << "event named" << eh.name;
572 switch (type) {
573 case top: top_event = eh; break;
574 case high: high_events.append(eh); break;
575 case mid: mid_events.append(eh); break;
576 case low: low_events.append(eh); break;
577 default: break;
578 }
579 break;
580 }
581 case bottom: {
582 BottomEvent eb;
583 eb.event.orderinpalette = orderinpalette;
584 eb.event.type = type;
585 QtParser::parseBottomEventChildren(r, eb);
586 if (!QtParser::hasValidEventName(r, eb.event.name))
587 return false;
588 qDebug() << "parsed Bottom event named" << eb.event.name;
589 bottom_events.append(eb);
590 break;
591 }
592 case sound: {
593 SpectrumEvent espec;
594 espec.orderinpalette = orderinpalette;
595 QtParser::parseSpectrumEventChildren(r, espec);
596 if (!QtParser::hasValidEventName(r, espec.name))
597 return false;
598 qDebug() << "parsed Spectrum event named" << espec.name;
599 spectrum_events.append(espec);
600 break;
601 }
602 case note: {
603 NoteEvent en;
604 en.orderinpalette = orderinpalette;
605 QtParser::parseNoteEventChildren(r, en);
606 if (!QtParser::hasValidEventName(r, en.name))
607 return false;
608 qDebug() << "parsed Note event named " << en.name;
609 note_events.append(en);
610 break;
611 }
612 case env: {
613 EnvelopeEvent ee;
614 ee.orderinpalette = orderinpalette;
615 ee.name = QtParser::nextRequiredChildInner(
616 r, QStringView(u"Name"));
617 ee.envelope_builder = QtParser::nextRequiredChildInner(
618 r, QStringView(u"EnvelopeBuilder"));
619 QtParser::consumeRest(r);
620 if (!QtParser::hasValidEventName(r, ee.name))
621 return false;
622 qDebug() << "parsed Envelope event named" << ee.name;
623 envelope_events.append(ee);
624 break;
625 }
626 case sieve: {
627 SieveEvent esi;
628 esi.orderinpalette = orderinpalette;
629 esi.name = QtParser::nextRequiredChildInner(
630 r, QStringView(u"Name"));
631 esi.sieve_builder = QtParser::nextRequiredChildInner(
632 r, QStringView(u"SieveBuilder"));
633 QtParser::consumeRest(r);
634 if (!QtParser::hasValidEventName(r, esi.name))
635 return false;
636 qDebug() << "parsed Sieve event named" << esi.name;
637 sieve_events.append(esi);
638 break;
639 }
640 case spa: {
641 SpaEvent espa;
642 espa.orderinpalette = orderinpalette;
643 espa.name = QtParser::nextRequiredChildInner(
644 r, QStringView(u"Name"));
645 espa.spatialization = QtParser::nextRequiredChildInner(
646 r, QStringView(u"Spatialization"));
647 QtParser::consumeRest(r);
648 if (!QtParser::hasValidEventName(r, espa.name))
649 return false;
650 qDebug() << "parsed Spa event named" << espa.name;
651 spa_events.append(espa);
652 break;
653 }
654 case pattern: {
655 PatternEvent ep;
656 ep.orderinpalette = orderinpalette;
657 ep.name = QtParser::nextRequiredChildInner(
658 r, QStringView(u"Name"));
659 ep.pattern_builder = QtParser::nextRequiredChildInner(
660 r, QStringView(u"PatternBuilder"));
661 QtParser::consumeRest(r);
662 if (!QtParser::hasValidEventName(r, ep.name))
663 return false;
664 qDebug() << "parsed Pattern event named" << ep.name;
665 pattern_events.append(ep);
666 break;
667 }
668 case reverb: {
669 ReverbEvent er;
670 er.orderinpalette = orderinpalette;
671 er.name = QtParser::nextRequiredChildInner(
672 r, QStringView(u"Name"));
673 er.reverberation = QtParser::nextRequiredChildInner(
674 r, QStringView(u"Reverberation"));
675 QtParser::consumeRest(r);
676 if (!QtParser::hasValidEventName(r, er.name))
677 return false;
678 qDebug() << "parsed Reverb event named" << er.name;
679 reverb_events.append(er);
680 break;
681 }
682 case filter: {
683 FilterEvent ef;
684 ef.orderinpalette = orderinpalette;
685 ef.name = QtParser::nextRequiredChildInner(
686 r, QStringView(u"Name"));
687 ef.filter_builder = QtParser::nextRequiredChildInner(
688 r, QStringView(u"FilterBuilder"));
689 QtParser::consumeRest(r);
690 if (!QtParser::hasValidEventName(r, ef.name))
691 return false;
692 qDebug() << "parsed Filter event named" << ef.name;
693 filter_events.append(ef);
694 break;
695 }
696 default:
697 return false;
698 }
699
700 if (parsedType)
701 *parsedType = type;
702 return true;
703}

705bool ProjectManager::parse(Project* p, const QString& filepath,
706 QString* errorMessage) {
707 const auto fail = [errorMessage](const QString& message) {
708 if (errorMessage)
709 *errorMessage = message;
710 return false;
711 };
712
713 QFile file(filepath);
714 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
715 return fail(QStringLiteral("Cannot read the project file: %1")

723 .arg(r.lineNumber())
724 .arg(r.columnNumber())
725 .arg(r.errorString()));
726 }
727 return fail(QStringLiteral("The file does not contain an XML root element."));
728 }

730 if (r.name() != QStringView(u"ProjectRoot")) {
731 return fail(QStringLiteral(
732 "Expected <ProjectRoot> at line %1, but found <%2>.")
733 .arg(r.lineNumber())
734 .arg(r.name().toString()));
735 }

737 bool foundProjectConfiguration = false;
738 bool foundEvents = false;
739 bool foundTopEvent = false;
740 QString configuredTopEventName;
741 QString structureError;
742
743 while (r.readNextStartElement()) {
744 const QString elName = r.name().toString();
745
746 if (elName == "ProjectConfiguration") {
747 if (foundProjectConfiguration) {
748 structureError = QStringLiteral(
749 "The project contains more than one <ProjectConfiguration> section.");
750 r.skipCurrentElement();
751 continue;
752 }
753 foundProjectConfiguration = true;
754
755 QSet<QString> requiredFields{
756 QStringLiteral("Title"), QStringLiteral("FileFlag"),
757 QStringLiteral("TopEvent"), QStringLiteral("PieceStartTime"),
758 QStringLiteral("Duration"), QStringLiteral("NumberOfStaff"),
759 QStringLiteral("NumberOfChannels"),
760 QStringLiteral("SampleRate"), QStringLiteral("SampleSize"),
761 QStringLiteral("NumberOfThreads")
762 };
763
764 // Older LASSIE versions omitted false boolean fields entirely.
765 p->synthesis = false;
766 p->score = false;
767 p->grand_staff = false;
768 p->output_particel = false;

770 const QString field = r.name().toString();
771 requiredFields.remove(field);
772
773 if (field == QStringLiteral("Title")) {
774 // The filename remains the authoritative in-app title.
775 QtParser::readInner(r);
776 } else if (field == QStringLiteral("FileFlag")) {
777 p->file_flag = QtParser::readInner(r);
778 } else if (field == QStringLiteral("TopEvent")) {
779 configuredTopEventName = QtParser::readInner(r).trimmed();
780 } else if (field == QStringLiteral("PieceStartTime")) {
781 p->start_time = QtParser::readInner(r);
782 } else if (field == QStringLiteral("Duration")) {
783 p->duration = QtParser::readInner(r);
784 } else if (field == QStringLiteral("Synthesis")) {
785 p->synthesis = (QtParser::readInner(r) == QStringLiteral("True"));
786 } else if (field == QStringLiteral("Score")) {
787 p->score = (QtParser::readInner(r) == QStringLiteral("True"));
788 } else if (field == QStringLiteral("GrandStaff")) {
789 p->grand_staff = (QtParser::readInner(r) == QStringLiteral("True"));
790 } else if (field == QStringLiteral("NumberOfStaff")) {
791 p->num_staffs = QtParser::readInner(r);
792 } else if (field == QStringLiteral("NumberOfChannels")) {
793 p->num_channels = QtParser::readInner(r);
794 } else if (field == QStringLiteral("SampleRate")) {
795 p->sample_rate = QtParser::readInner(r);
796 } else if (field == QStringLiteral("SampleSize")) {
797 p->sample_size = QtParser::readInner(r);
798 } else if (field == QStringLiteral("NumberOfThreads")) {
799 p->num_threads = QtParser::readInner(r);
800 } else if (field == QStringLiteral("OutputParticel")) {
801 p->output_particel =
802 (QtParser::readInner(r) == QStringLiteral("True"));
803 } else if (field == QStringLiteral("Seed")) {
804 p->seed = QtParser::readInner(r);
805 } else {
806 r.skipCurrentElement();
807 }
808 }
809

811 QStringList missingFields = requiredFields.values();
812 missingFields.sort();
813 structureError = QStringLiteral(
814 "<ProjectConfiguration> is missing required field(s): %1.")
815 .arg(missingFields.join(QStringLiteral(", ")));
816 }
817 }
818 else if (elName == "NoteModifiers") {
819 // First child: default modifiers bitstring (currently ignored — see
820 // pre-Qt code: the loop that consumed it was commented out).
821 if (r.readNextStartElement()) {
822 QtParser::readInner(r);
823 // Second child: custom modifiers list
824 if (r.readNextStartElement()) {
825 while (r.readNextStartElement())
826 p->custom_note_modifiers.append(QtParser::readInner(r));
827 }
828 QtParser::consumeRest(r);
829 }
830 qDebug() << "Passed modifiers";
831 }
832 else if (elName == "EnvelopeLibrary") {
833 QString envLibText = r.readElementText(QXmlStreamReader::IncludeChildElements);
834 EnvelopeLibrary* envelopeLibrary = new EnvelopeLibrary();
835 if (!envLibText.isEmpty()) {
836 QString tempPath = filepath + ".lib.temp";
837 QFile temp(tempPath);
838 if (temp.open(QIODevice::WriteOnly | QIODevice::Text)) {
839 QTextStream out(&temp);
840 out << envLibText;
841 temp.close();
842 envelopeLibrary->loadLibraryNewFormat(
843 const_cast<char*>(tempPath.toLocal8Bit().constData()));

845 }
846 }
847 EnvelopeLibraryEntry* previousEntry = nullptr;
848 for (int i = 1; i <= envelopeLibrary->size(); i++) {
849 Envelope* thisEnvelope = envelopeLibrary->getEnvelope(i);
850 EnvelopeLibraryEntry* thisEntry = new EnvelopeLibraryEntry(thisEnvelope, i);
851 delete thisEnvelope;

853 p->elentry = thisEntry;
854 thisEntry->prev = nullptr;
855 } else {
856 previousEntry->next = thisEntry;
857 thisEntry->prev = previousEntry;
858 }
859 previousEntry = thisEntry;
860 }
861 delete envelopeLibrary;
862 qDebug() << "Passed envelopes";
863 }
864 else if (elName == "MarkovModelLibrary") {
865 QString data = r.readElementText(QXmlStreamReader::IncludeChildElements);
866 if (!data.isEmpty()) {

869 ss >> size;
870 p->markov_models.resize(size);
871 std::string line;
872 std::getline(ss, line, '\n');
873 for (long long i = 0; i < size; i++) {
874 std::string modelText;
875 std::getline(ss, line, '\n'); modelText = line + '\n';
876 std::getline(ss, line, '\n'); modelText += line + '\n';
877 std::getline(ss, line, '\n'); modelText += line + '\n';
878 std::getline(ss, line, '\n'); modelText += line;
879 p->markov_models[i].from_str(modelText);
880 }
881 }
882 qDebug() << "Passed markov";
883 }
884 else if (elName == "Events") {
885 if (foundEvents) {
886 structureError = QStringLiteral(
887 "The project contains more than one <Events> section.");
888 r.skipCurrentElement();
889 continue;
890 }
891 foundEvents = true;
892 while (r.readNextStartElement()) {
893 if (r.name() == QStringView(u"Event")) {
894 Eventtype parsedType = top;
895 if (p->parseEvent(r, &parsedType) && parsedType == top) {
896 if (foundTopEvent) {
897 structureError = QStringLiteral(
898 "The project contains more than one top event.");
899 }
900 foundTopEvent = true;
901 }
902 } else {
903 r.skipCurrentElement();
904 }
905 }
906 }
907 else {
908 r.skipCurrentElement();
909 }
910 }
911
912 if (r.hasError()) {
913 return fail(QStringLiteral("XML error at line %1, column %2: %3")
914 .arg(r.lineNumber())
915 .arg(r.columnNumber())
916 .arg(r.errorString()));
917 }
918 if (!foundProjectConfiguration)
919 return fail(QStringLiteral("The project is missing <ProjectConfiguration>."));
920 if (!foundEvents)
921 return fail(QStringLiteral("The project is missing <Events>."));
922 if (!foundTopEvent)
923 return fail(QStringLiteral("The project is missing a top event."));
924 if (!structureError.isEmpty())
925 return fail(structureError);
926 if (configuredTopEventName.isEmpty()) {
927 return fail(QStringLiteral(
928 "<ProjectConfiguration><TopEvent> must name the top event."));
929 }
930 if (p->top_event.name != configuredTopEventName) {
931 return fail(QStringLiteral(
932 "Configured top event '%1' does not match parsed top event '%2'.")
933 .arg(configuredTopEventName, p->top_event.name));
934 }
935
936 if (errorMessage)
937 errorMessage->clear();
938 return true;
939}
